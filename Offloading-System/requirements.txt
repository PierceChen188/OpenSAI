# --- Redis client: MUST stay on the 5.0.x line ---
# redis-py >= 6 negotiates RESP3 with a `HELLO` command that Redis server 5.x
# rejects ("unknown command HELLO"). Because RedisUtils swallows RedisError,
# every cache op then fails *silently* (devices never register, tasks never
# dispatch). Keep this pin compatible with your Redis server version.
redis==5.0.8

# --- Web frontend (dag_offload/user_client.py) ---
# NOTE: flask-cors was MISSING from the original requirements even though
# user_client.py imports it (`from flask_cors import CORS`).
Flask==3.1.3
flask-cors==6.0.2

# --- Device monitoring / resource limiting ---
psutil==7.2.2
PyYAML==6.0.3
# threadpoolctl: align BLAS/OpenMP thread count to the per-task CPU-core budget (B2).
# Without it, cpu_affinity pins cores but numpy/torch still spawn ~ncores threads that
# thrash the pinned subset (matmul gets SLOWER, not faster). Optional: run_task falls back
# to affinity-only if it's missing, but per-task core allocation won't speed up BLAS compute.
threadpoolctl==3.6.0

# --- Data plane: device-to-device bulk transfer over QUIC (Phase 2a) ---
# Used by network/dataplane/ (ArtifactServer + fetch_blob). Pulls in `cryptography`.
aioquic==1.3.0
